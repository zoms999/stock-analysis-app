기획서 v1.1과 논의된 관리자 기능을 모두 포함한 **최종 통합 SQL 스크립트**입니다.

이 스크립트를 Supabase 대시보드의 **SQL Editor**에 그대로 복사해서 실행(`Run`)하시면, 데이터베이스 구축이 한 번에 완료됩니다.

---

### [통합 SQL 스크립트] 차트 기반 투자 커뮤니티 & 리워드 플랫폼

**순서:** Enum 정의 -> 테이블 생성 -> 인덱스 설정 -> 트리거(자동 회원가입) 설정 -> RLS(보안) 기초 설정

```sql
-- 1. ENUM 타입 정의 (상태값 관리)
CREATE TYPE point_type AS ENUM ('EARN', 'USE', 'WITHDRAW', 'ADMIN', 'REFUND');
CREATE TYPE tournament_status AS ENUM ('OPEN', 'ACTIVE', 'CLOSED', 'SETTLED');
CREATE TYPE tournament_type AS ENUM ('KOSPI_DECIMAL', 'STOCK_PREDICTION');

-- 2. 회원 프로필 (Profiles) - Auth 및 관리자 기능 통합
CREATE TABLE public.profiles (
  id UUID REFERENCES auth.users(id) ON DELETE CASCADE PRIMARY KEY, -- Supabase Auth 연결
  email TEXT NOT NULL,
  nickname TEXT UNIQUE,
  avatar_url TEXT,
  
  -- 등급 및 권한
  user_level INT DEFAULT 1 CHECK (user_level BETWEEN 1 AND 10), -- 1~10등급
  is_partner BOOLEAN DEFAULT FALSE, -- 파트너 여부
  is_admin BOOLEAN DEFAULT FALSE,   -- 관리자 여부 (★추가됨)
  admin_memo TEXT,                  -- 관리자용 유저 메모 (★추가됨)
  
  -- 마케팅/추천인
  referral_code TEXT UNIQUE,
  referred_by UUID REFERENCES public.profiles(id),
  
  -- 자산 (포인트)
  point_balance INT DEFAULT 0,
  
  -- 설정
  language_code VARCHAR(10) DEFAULT 'ko',
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- 3. 관리자 감사 로그 (Admin Logs)
CREATE TABLE public.admin_logs (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  admin_id UUID REFERENCES public.profiles(id) NOT NULL,
  target_user_id UUID REFERENCES public.profiles(id),
  action_type VARCHAR(50) NOT NULL, -- 예: 'APPROVE_WITHDRAW', 'BAN_USER'
  details JSONB, -- 변경 내역 상세
  ip_address TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 4. 구독 플랜 (Plans)
CREATE TABLE public.plans (
  id SERIAL PRIMARY KEY,
  name VARCHAR(50) NOT NULL, -- Free, Light, Standard, Premium, VIP
  price DECIMAL(10, 2) NOT NULL,
  target_level INT NOT NULL, -- 결제 시 부여될 등급
  
  -- 제한 설정
  daily_view_limit INT DEFAULT 3,
  daily_write_limit INT DEFAULT 5,
  access_max_level INT DEFAULT 5,
  
  stripe_product_id TEXT -- Stripe 연동용
);

-- 5. 구독 현황 (Subscriptions)
CREATE TABLE public.subscriptions (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id UUID REFERENCES public.profiles(id) NOT NULL,
  plan_id INT REFERENCES public.plans(id),
  
  status VARCHAR(20) DEFAULT 'active',
  stripe_subscription_id TEXT,
  
  current_period_start TIMESTAMPTZ,
  current_period_end TIMESTAMPTZ,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 6. 게시글 및 차트 (Posts)
CREATE TABLE public.posts (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id UUID REFERENCES public.profiles(id) NOT NULL,
  
  title TEXT NOT NULL,
  content TEXT,
  
  -- 차트 데이터
  ticker_symbol VARCHAR(50), -- 예: BTC/USDT
  chart_config JSONB,        -- 차트 설정 및 드로잉 데이터
  
  required_level INT DEFAULT 1,
  view_count INT DEFAULT 0,
  
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 7. 활동 로그 (Activity Logs - 일일 제한 체크용)
CREATE TABLE public.activity_logs (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  user_id UUID REFERENCES public.profiles(id) NOT NULL,
  type VARCHAR(20) CHECK (type IN ('VIEW', 'WRITE')),
  target_post_id UUID REFERENCES public.posts(id),
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 8. 포인트 내역 (Point Transactions)
CREATE TABLE public.point_transactions (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  user_id UUID REFERENCES public.profiles(id) NOT NULL,
  
  amount INT NOT NULL,
  transaction_type point_type NOT NULL,
  description TEXT,
  
  related_post_id UUID REFERENCES public.posts(id),
  related_event_id UUID, -- 토너먼트 ID는 나중에 연결 (순환 참조 방지)
  
  is_approved BOOLEAN DEFAULT TRUE, -- 관리자 승인 필요 시 False
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 9. 토너먼트 (Tournaments)
CREATE TABLE public.tournaments (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  title TEXT NOT NULL,
  description TEXT,
  
  event_type tournament_type NOT NULL,
  target_date TIMESTAMPTZ NOT NULL,
  
  status tournament_status DEFAULT 'OPEN',
  prize_pool TEXT,
  
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 10. 토너먼트 참가 (Tournament Entries - 바이럴 로직 포함)
CREATE TABLE public.tournament_entries (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  tournament_id UUID REFERENCES public.tournaments(id) NOT NULL,
  user_id UUID REFERENCES public.profiles(id) NOT NULL,
  
  -- 예측 값
  prediction_value DECIMAL,
  prediction_json JSONB,
  
  -- 패자부활 로직
  is_eliminated BOOLEAN DEFAULT FALSE,
  re_entry_count INT DEFAULT 0,
  max_re_entry INT DEFAULT 3,
  
  created_at TIMESTAMPTZ DEFAULT NOW(),
  UNIQUE(tournament_id, user_id)
);

-- 11. 파트너 정산 (Partner Settlements)
CREATE TABLE public.partner_settlements (
  id SERIAL PRIMARY KEY,
  partner_id UUID REFERENCES public.profiles(id) NOT NULL,
  source_user_id UUID REFERENCES public.profiles(id),
  
  payment_amount DECIMAL(10, 2),
  commission_rate DECIMAL(5, 2),
  settlement_amount DECIMAL(10, 2),
  
  is_paid BOOLEAN DEFAULT FALSE,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 12. 성능 최적화를 위한 인덱스 생성
CREATE INDEX idx_profiles_referral ON public.profiles(referral_code);
CREATE INDEX idx_posts_user ON public.posts(user_id);
CREATE INDEX idx_posts_ticker ON public.posts(ticker_symbol);
CREATE INDEX idx_activity_logs_daily ON public.activity_logs(user_id, created_at);
CREATE INDEX idx_points_user ON public.point_transactions(user_id);
CREATE INDEX idx_entries_tournament ON public.tournament_entries(tournament_id);

-- 13. [필수] 회원가입 시 Profile 자동 생성 트리거
CREATE OR REPLACE FUNCTION public.handle_new_user()
RETURNS TRIGGER AS $$
BEGIN
  INSERT INTO public.profiles (id, email, nickname)
  VALUES (
    new.id, 
    new.email, 
    COALESCE(new.raw_user_meta_data->>'full_name', new.raw_user_meta_data->>'name', 'User_' || substr(new.id::text, 1, 8))
  );
  RETURN new;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- 트리거 연결 (기존 트리거가 있다면 삭제 후 재생성)
DROP TRIGGER IF EXISTS on_auth_user_created ON auth.users;
CREATE TRIGGER on_auth_user_created
  AFTER INSERT ON auth.users
  FOR EACH ROW EXECUTE PROCEDURE public.handle_new_user();

-- 14. [보안] RLS(Row Level Security) 활성화
-- 모든 테이블에 대해 RLS 활성화 (기본적으로 모든 접근 차단됨)
ALTER TABLE public.profiles ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.posts ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.point_transactions ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.subscriptions ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.tournaments ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.tournament_entries ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.partner_settlements ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.admin_logs ENABLE ROW LEVEL SECURITY;

-- [기본 정책 예시] Profiles: 누구나 조회는 가능(닉네임 등), 수정은 본인만
CREATE POLICY "Public profiles are viewable by everyone" ON public.profiles FOR SELECT USING (true);
CREATE POLICY "Users can update own profile" ON public.profiles FOR UPDATE USING (auth.uid() = id);

-- [기본 정책 예시] Posts: 누구나 조회 가능 (상세 로직은 백엔드에서 등급 체크 권장)
CREATE POLICY "Posts are viewable by everyone" ON public.posts FOR SELECT USING (true);
CREATE POLICY "Users can create posts" ON public.posts FOR INSERT WITH CHECK (auth.uid() = user_id);
```

### 실행 후 확인해야 할 사항

1.  **관리자 계정 만들기**:
    *   일단 일반 회원가입(소셜 로그인 등)을 진행합니다.
    *   Supabase 대시보드 -> Table Editor -> `profiles` 테이블로 이동합니다.
    *   방금 가입한 본인 계정의 `is_admin` 컬럼 값을 `TRUE`로 변경합니다.
2.  **구독 플랜 데이터 넣기**:
    *   `plans` 테이블은 비어있으므로, SQL Editor에서 초기 데이터를 넣어줍니다.
    ```sql
    INSERT INTO public.plans (name, price, target_level, daily_view_limit, daily_write_limit) VALUES

    ```
